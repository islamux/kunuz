Arabic search pipeline

// verbatim from src/utils/arabic.ts:4-17
export function removeTashkeel(text: string): string {
  if (!text) return '';
  return text
    .replace(/\u0640/g, '')           // tatweel (kashida)
    .replace([\u064B-\u065F\u0670]/g, '') // harakat
    .replace([إأآٱ]/g, 'ا')          // alef variants → alef
    .replace(ة/g, 'ه')               // teh marbuta → heh
    .replace(ى/g, 'ي');              // alef maqsura → yeh
}
// verbatim from src/utils/arabic.ts:22-27
export function matchesSearch(content: string, query: string): boolean {
  if (!query.trim()) return true;
  const cleanContent = removeTashkeel(content.toLowerCase());
  const cleanQuery = removeTashkeel(query.toLowerCase().trim());
  return cleanContent.includes(cleanQuery);
}

Daily treasure determinism

// verbatim from src/data/index.ts:48-55
export function getDailyTreasure(): Treasure {
  const today = new Date();
  const dayOfYear = Math.floor(
    (today.getTime() - new Date(today.getFullYear(), 0, 0).getTime())
    / 1000 / 60 / 60 / 24
  );
  const index = Math.abs(dayOfYear) % ALL_TREASURES.length;
  return ALL_TREASURES[index];
}
  • Math: dayOfYear % 140 — deterministic for the same calendar day
  • Test-verified: data.test.ts:89–91 asserts getDailyTreasure().id === getDailyTreasure().id
  • Timezone caveat: uses local new Date() — the "same" treasure may differ across timezones at day boundaries
